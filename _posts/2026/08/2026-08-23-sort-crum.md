## クラムソートを使用する

クラムソート (`crumsort`) は[フラックスソート](/2026/08/18/sort-flux.html)と同じく、ピボットで分割して再帰するトップダウン方式のハイブリッド比較ソートである。違いは分割が不安定かつインプレースな点で、補助配列への二重書き込みの代わりにフルクラム（支点）分割を使う。

整列度が高い区間や小区間・不均衡時は[クワッドソート](/2026/08/17/sort-quad.html)（マージソート系）へ切り替える。本番実装は分岐の少ない比較や固定長の小さなスワップ領域（既定で数百要素）を用いるが、本記事の実装は説明用に簡略化している。小区間・フォールバックは[挿入ソート](/2026/05/05/sort-insertion.html)／[マージソート](/2026/05/03/sort-merge.html)で代用し、ピボットは9点の準中央値のみとする。

1.  **アナライザ**: 全体が昇順なら何もしない。降順（同値を含む非増加）なら反転して終了する（比較・移動とも `O(n)`）。配列を 4 分割し、各区間の隣接昇順ペアが半数超ならその区間をマージソートで仕上げる（本番ではクワッドソート）。
2.  **ピボット選択**: 区間をほぼ等間隔に 9 点取り、3 組の三点中央値の中央値（準中央値）をピボットにする。
3.  **フルクラム分割**: ピボット値を 1 要素ぶんの退避スロットに置き、先頭側（head）と末尾側（tail）から走査する。`≤ ピボット` は前方へ、`> ピボット` は後方へ 2 代入で書き分ける（通常の 3 代入スワップより軽い）。同値の相対順序は保たない。
4.  **等値の第二走査**: 右側が空（すべて `≤ ピボット`）なら、`< ピボット` だけを前方へ寄せて等値帯を再帰から外す。重複の多い入力向けの対策である。
5.  **不均衡フォールバック**: 左右の長さ比が 1:16 を超えて偏ったら、両側をマージソートする（本番ではクワッドソート）。最悪計算量を `O(n log n)` に抑えるためのガードである。
6.  **小区間**: 要素数が閾値未満なら挿入ソートで仕上げる（本番の閾値付近ではクワッドソートの小区間ルーチン）。

```pseudocode
procedure crum_fulcrum_partition(A, pivot)
  // A[0] にピボットを置き、値はローカルへ退避（1 要素の swap 領域）
  move pivot into A[0]; pivot_val := A[0]
  head := 0; tail := length(A) - 1
  loop
    while head < tail and A[tail] > pivot_val
      tail := tail - 1
    if head ≥ tail then
      A[head] := pivot_val; return head
    A[head] := A[tail]; head := head + 1
    while head < tail and A[head] ≤ pivot_val
      head := head + 1
    if head ≥ tail then
      A[head] := pivot_val; return head
    A[tail] := A[head]; tail := tail - 1

procedure crum_sort_range(A)
  if length(A) < INSERTION_THRESHOLD then
    insertion_sort(A); return
  pivot := quasimedian_of_9(A)
  mid := crum_fulcrum_partition(A, pivot)
  left := mid; right := length(A) - mid - 1
  if right = 0 then
    gather keys < pivot to front
    crum_sort_range(A[0 .. lt))
    return
  if left < length(A)/16 or right < length(A)/16 then
    merge_sort(A[0 .. left)); merge_sort(A[mid+1 .. end)); return
  crum_sort_range(A[0 .. left))
  crum_sort_range(A[mid+1 .. end))

procedure crumsort(A)
  if A is sorted then return
  if A is reverse-sorted then reverse(A); return
  for each quarter Q of A
    if ordered_pairs(Q) > half then merge_sort(Q)
  if A is sorted then return
  crum_sort_range(A)
```

最良は整列済み検出により `O(n)`、平均・最悪は `O(n log n)` である。分割そのものはインプレースで、本番の補助メモリは小さな固定領域に抑えられる（不安定ソート）。説明用実装ではマージフォールバック用に最大 `O(n)` の作業領域を使う。

以下のデモでは視認性のため挿入閾値を 4、不均衡判定を 1/4 に緩めている。

  const INSERTION_THRESHOLD = 4;
  const UNBALANCE_RATIO = 4;

  function isSorted(a) {
    for (let i = 1; i < a.length; i++) {
      if (a[i - 1] > a[i]) return false;
    }
    return true;
  }

  function isReverseSorted(a) {
    for (let i = 1; i < a.length; i++) {
      if (a[i - 1] < a[i]) return false;
    }
    return true;
  }

  function orderedPairs(a, lo, hi) {
    let c = 0;
    for (let i = lo + 1; i <= hi; i++) {
      if (a[i - 1] <= a[i]) c++;
    }
    return c;

  function median3Idx(a, i, j, k) {
    const x = a[i];
    const y = a[j];
    const z = a[k];
    if (x < y) {
      if (y < z) return j;
      if (x < z) return k;
      return i;
    }
    if (x < z) return i;
    if (y < z) return k;
    return j;

    const n = hi - lo + 1;
    if (n < 9) return a[lo + Math.floor(n / 2)];

    const i0 = lo;
    const i1 = lo + step;

    const m2 = median3Idx(a, i6, i7, i8);
    return a[median3Idx(a, m0, m1, m2)];

    const a = initial.slice();
    const steps = [];
    const n = a.length;

    function pushPhase(text) {
      steps.push({
        kind: 'phase',
        text: text,
        arr: a.slice(),

    function insertionSort(lo, hi) {

            a[j - 1] = a[j];
            a[j] = t;
            steps.push({
              kind: 'swap',
              lo: j - 1,
              hi: j,
              arr: a.slice(),
              phase: 'insert',
            });
            j--;
          } else {
            break;
          }
        }
      }
    }

    function mergeSort(lo, hi) {
      if (hi <= lo) return;
      if (hi - lo + 1 < INSERTION_THRESHOLD) {
        insertionSort(lo, hi);
        return;
      }
      const mid = lo + Math.floor((hi - lo) / 2);
      mergeSort(lo, mid);
      mergeSort(mid + 1, hi);
      const left = a.slice(lo, mid + 1);
      const right = a.slice(mid + 1, hi + 1);
      let i = 0;
      let j = 0;
      let k = lo;
      pushPhase(`マージ: 位置 ${lo} … ${hi}`);
      while (i < left.length && j < right.length) {
        steps.push({
          kind: 'compare',
          lo: lo + i,
          hi: mid + 1 + j,
          arr: a.slice(),
          phase: 'merge',
        });
        if (left[i] <= right[j]) {
          a[k] = left[i];
          steps.push({
            kind: 'write',
            idx: k,
            arr: a.slice(),
            phase: 'merge',
          });
          i++;
        } else {
          a[k] = right[j];
          steps.push({
            kind: 'write',
            idx: k,
            arr: a.slice(),
            phase: 'merge',
          });
          j++;
        }
        k++;
      }
      while (i < left.length) {
        a[k] = left[i];
        steps.push({
          kind: 'write',
          idx: k,
          arr: a.slice(),
          phase: 'merge',
        });
        i++;
        k++;
      }
      while (j < right.length) {
        a[k] = right[j];
        steps.push({
          kind: 'write',
          idx: k,
          arr: a.slice(),
          phase: 'merge',
        });
        j++;
        k++;
      }
    }

    function fulcrumPartition(lo, hi, pivot) {
      let pivotIdx = lo;
      for (let i = lo; i <= hi; i++) {
        if (a[i] === pivot) {
          pivotIdx = i;
          break;
        }
      }
      if (pivotIdx !== lo) {
        const t = a[lo];
        a[lo] = a[pivotIdx];
        a[pivotIdx] = t;
        steps.push({
          kind: 'swap',
          lo: lo,
          hi: pivotIdx,
          arr: a.slice(),
          phase: 'pivot',
        });
      }

      const pivotVal = a[lo];
      let head = lo;
      let tail = hi;
      pushPhase(`ピボット ${pivotVal} で位置 ${lo} … ${hi} をフルクラム分割`);

      while (true) {
        while (head < tail && a[tail] > pivotVal) {
          steps.push({
            kind: 'compare',
            lo: tail,
            hi: tail,
            pivotVal: pivotVal,
            arr: a.slice(),
            phase: 'part',
          });
          tail--;
        }
        if (head < tail) {
          steps.push({
            kind: 'compare',
            lo: tail,
            hi: tail,
            pivotVal: pivotVal,
            arr: a.slice(),
            phase: 'part',
          });

          a[head] = pivotVal;
          steps.push({
            kind: 'write',
            idx: head,
            arr: a.slice(),
            phase: 'pivot_home',
          });
          return head;
        }
        a[head] = a[tail];
        steps.push({
          kind: 'part_progress',
          writeIdx: head,
          fromIdx: tail,
          side: 'left',
          arr: a.slice(),
          pivotVal: pivotVal,
        });
        head++;

        while (head < tail && a[head] <= pivotVal) {
          steps.push({
            kind: 'compare',
            lo: head,
            hi: head,
            pivotVal: pivotVal,
            arr: a.slice(),
            phase: 'part',
          });
          head++;
        }
        if (head < tail) {
          steps.push({
            kind: 'compare',
            lo: head,
            hi: head,
            pivotVal: pivotVal,
            arr: a.slice(),
            phase: 'part',
          });
        }
        if (head >= tail) {
          a[head] = pivotVal;
          steps.push({
            kind: 'write',
            idx: head,
            arr: a.slice(),
            phase: 'pivot_home',
          });
          return head;

        steps.push({
          kind: 'part_progress',
          writeIdx: tail,
          fromIdx: head,
          side: 'right',
          arr: a.slice(),
          pivotVal: pivotVal,
        });
        tail--;
      }

    function crumRange(lo, hi) {
      const size = hi - lo + 1;
      if (size <= 1) return;
      if (size < INSERTION_THRESHOLD) {
        insertionSort(lo, hi);
        return;
      }

      const pivot = quasimedian9(a, lo, hi);
      const mid = fulcrumPartition(lo, hi, pivot);
      const left = mid - lo;
      const right = hi - mid;

      steps.push({
        kind: 'part_end',
        lo: lo,
        hi: hi,
        split: mid,
        pivotVal: pivot,
        arr: a.slice(),
      });

      if (right === 0) {
        let lt = 0;
        for (let i = lo; i <= hi; i++) {
          if (a[i] < pivot) {
            const dest = lo + lt;
            if (dest !== i) {
              const t = a[dest];
              a[dest] = a[i];
              a[i] = t;
              steps.push({
                kind: 'swap',
                lo: dest,
                hi: i,
                arr: a.slice(),
                phase: 'equals',
              });
            }
            lt++;
          }
        }
        pushPhase(
          `第二走査: 等値 ${pivot} を確定し、厳密に小さい側だけ再帰`
        );
        if (lt > 1) crumRange(lo, lo + lt - 1);
        return;
      }

      if (left < size / UNBALANCE_RATIO || right < size / UNBALANCE_RATIO) {
        pushPhase(`不均衡のためマージソートへ切替: 位置 ${lo} … ${hi}`);
        if (left > 1) mergeSort(lo, mid - 1);
        if (right > 1) mergeSort(mid + 1, hi);
        return;
      }

      if (left > 1) crumRange(lo, mid - 1);
      if (right > 1) crumRange(mid + 1, hi);
    }

    if (n === 0) {
      steps.push({ kind: 'done', arr: [] });
      return steps;
    }

    pushPhase('アナライザ: 整列済み／逆順／区間の整列度を検査');
    if (isSorted(a)) {
      pushPhase('全体が昇順のため終了');
      steps.push({ kind: 'done', arr: a.slice() });
      return steps;
    }
    if (isReverseSorted(a)) {
      pushPhase('全体が降順のため反転');
      let lo = 0;
      let hi = n - 1;
      while (lo < hi) {
        const t = a[lo];
        a[lo] = a[hi];
        a[hi] = t;
        steps.push({
          kind: 'swap',
          lo: lo,
          hi: hi,
          arr: a.slice(),
          phase: 'reverse',
        });
        lo++;
        hi--;
      }
      steps.push({ kind: 'done', arr: a.slice() });
      return steps;
    }

    const q = Math.floor(n / 4);
    if (q >= 2) {
      const bounds = [0, q, q * 2, q * 3, n];
      for (let s = 0; s < 4; s++) {
        const lo = bounds[s];
        const hi = bounds[s + 1] - 1;
        if (hi - lo < 1) continue;
        const pairs = hi - lo;
        if (orderedPairs(a, lo, hi) * 2 > pairs) {
          pushPhase(`区間 ${lo} … ${hi} の整列度が高いためマージソート`);
          mergeSort(lo, hi);
        }
      }
      if (isSorted(a)) {
        steps.push({ kind: 'done', arr: a.slice() });
        return steps;
      }
    }

    crumRange(0, n - 1);
    steps.push({ kind: 'done', arr: a.slice() });
    return steps;
  }

## 類似アルゴリズムとの相違点

[フラックスソート](/2026/08/18/sort-flux.html)は安定な二重書き込み分割と最大 `O(n)` の補助配列を使う。クラムソートはフルクラム分割で不安定・インプレース寄りにし、ランダム入力では大規模でフラックスソートを追い抜きやすい一方、既存の並びパターンを崩しやすい。

[クワッドソート](/2026/08/17/sort-quad.html)は分割を行わず、ボトムアップのクワッドマージだけで完結する。クラムソートはランダム寄りの入力ではフルクラム分割を主とし、整列度が高い区間・小区間・不均衡時だけクワッドソート系へ寄せる。

[ホア分割型クイックソート](/2026/08/14/sort-quick-hoare.html)も両端から詰めるインプレース分割だが、フルクラムはピボットを 1 要素の退避スロットに置き、3 代入のスワップを 2 代入の書き分けに置き換える。

[パターン撃退型クイックソート](/2026/07/27/sort-pattern-defeating-quick.html)も悪パターン対策のハイブリッドで不安定・低補助メモリである。クラムソートは先頭アナライザと不均衡時のマージ切替、等値の第二走査を組み合わせる点が近い。
