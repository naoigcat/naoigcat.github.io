## フラックスソートを使用する

フラックスソート (`fluxsort`) は[クイックソート](/2026/05/02/sort-quick-lumoto.html)と同じようにピボットで分割して再帰していくトップダウン方式で、二重書き込みによって安定に分割するのを主としたハイブリッドな比較ソートである。

整列度が高い区間や小区間・不均衡時は[クワッドソート](/2026/08/17/sort-quad.html)（マージソート系）へ切り替えるようになっており、分割時の分岐予測ミスを抑えやすい点と、先頭のアナライザで昇順・降順・区間の整列度を見て適応する点が特徴である。

本記事の実装は説明用に簡略化している。本番では小区間やフォールバックにクワッドソートを使うが、ここでは[挿入ソート](/2026/05/05/sort-insertion.html)／[マージソート](/2026/05/03/sort-merge.html)で代用する。ピボットも準中央値（9）のみとし、大規模向けの立方根近似中央値は扱わない。

1.  **アナライザ**: 全体が昇順なら何もしない。降順（同値を含む非増加）なら反転して終了する（比較・移動とも `O(n)`）。配列を 4 分割し、各区間の隣接昇順ペアが半数超ならその区間をマージソートで仕上げる（本番ではクワッドソート）。
2.  **ピボット選択**: 区間をほぼ等間隔に 9 点取り、3 組の三点中央値の中央値（準中央値）をピボットにする。
3.  **安定な二重書き込み分割**: 要素を先頭から走査し、`≤ ピボット` を本配列前方へ、`> ピボット` を後方（本番では swap 領域）へ、出現順を保ったまま書き分ける。
4.  **等値の第二走査**: 右側が空（すべて `≤ ピボット`）なら、`< ピボット` と `= ピボット` に分け直し、等値帯を再帰から外す。重複の多い入力向けの対策である。
5.  **不均衡フォールバック**: 左右の長さ比が 1:16 を超えて偏ったら、両側をマージソートする（本番ではクワッドソート）。最悪計算量を `O(n log n)` に抑えるためのガードである。
6.  **小区間**: 要素数が閾値未満なら挿入ソートで仕上げる（本番の閾値付近ではクワッドソートの小区間ルーチン）。

```pseudocode
procedure flux_stable_partition(A, pivot)
  L := empty; R := empty
  for each x in A in order
    if x ≤ pivot then append x to L else append x to R
  A := L concatenated with R
  return length(L)

procedure flux_sort_range(A)
  if length(A) < INSERTION_THRESHOLD then
    insertion_sort(A); return
  pivot := quasimedian_of_9(A)
  left := flux_stable_partition(A, pivot)
  right := length(A) - left
  if right = 0 then
    move keys < pivot to front, equals after them
    flux_sort_range(A[0 .. lt))
    return
  if left < length(A)/16 or right < length(A)/16 then
    merge_sort(A[0 .. left)); merge_sort(A[left .. end)); return
  flux_sort_range(A[0 .. left))
  flux_sort_range(A[left .. end))

procedure fluxsort(A)
  if A is sorted then return
  if A is reverse-sorted then reverse(A); return
  for each quarter Q of A
    if ordered_pairs(Q) > half then merge_sort(Q)
  if A is sorted then return
  flux_sort_range(A)
```

最良は整列済み検出により `O(n)`、平均・最悪は `O(n log n)` である。補助配列に最大 `O(n)` を使う安定ソートである。

以下のデモでは視認性のため挿入閾値を 4、不均衡判定を 1/4 に緩めている。

<script>
window.DemoSort && DemoSort.boot('flux-sort-demo', function (root) {
  /** デモ用に小区間閾値と不均衡比を緩め、分割／フォールバックが見えやすくしている。 */
  const INSERTION_THRESHOLD = 4;
  const UNBALANCE_RATIO = 4;

  function isSorted(a) {
    for (let i = 1; i < a.length; i++) {
      if (a[i - 1] > a[i]) return false;
    }
    return true;
  }

  function isReverseSorted(a) {
    for (let i = 1; i < a.length; i++) {
      if (a[i - 1] < a[i]) return false;
    }
    return true;
  }

  function orderedPairs(a, lo, hi) {
    let c = 0;
    for (let i = lo + 1; i <= hi; i++) {
      if (a[i - 1] <= a[i]) c++;
    }
    return c;
  }

  function median3Idx(a, i, j, k) {
    const x = a[i];
    const y = a[j];
    const z = a[k];
    if (x < y) {
      if (y < z) return j;
      if (x < z) return k;
      return i;
    }
    if (x < z) return i;
    if (y < z) return k;
    return j;
  }

  function quasimedian9(a, lo, hi) {
    const n = hi - lo + 1;
    if (n < 9) return a[lo + Math.floor(n / 2)];
    const step = Math.floor(n / 8);
    const i0 = lo;
    const i1 = lo + step;
    const i2 = lo + step * 2;
    const i3 = lo + step * 3;
    const i4 = lo + step * 4;
    const i5 = lo + step * 5;
    const i6 = lo + step * 6;
    const i7 = lo + step * 7;
    const i8 = hi;
    const m0 = median3Idx(a, i0, i1, i2);
    const m1 = median3Idx(a, i3, i4, i5);
    const m2 = median3Idx(a, i6, i7, i8);
    return a[median3Idx(a, m0, m1, m2)];
  }

  function generateSteps(initial) {
    const a = initial.slice();
    const steps = [];
    const n = a.length;

    function pushPhase(text) {
      steps.push({
        kind: 'phase',
        text: text,
        arr: a.slice(),
      });
    }

    function insertionSort(lo, hi) {
      pushPhase(`挿入ソート: 位置 ${lo} … ${hi}`);
      for (let i = lo + 1; i <= hi; i++) {
        let j = i;
        while (j > lo) {
          steps.push({
            kind: 'compare',
            lo: j - 1,
            hi: j,
            arr: a.slice(),
            phase: 'insert',
          });
          if (a[j - 1] > a[j]) {
            const t = a[j - 1];
            a[j - 1] = a[j];
            a[j] = t;
            steps.push({
              kind: 'swap',
              lo: j - 1,
              hi: j,
              arr: a.slice(),
              phase: 'insert',
            });
            j--;
          } else {
            break;
          }
        }
      }
    }

    function mergeSort(lo, hi) {
      if (hi <= lo) return;
      if (hi - lo + 1 < INSERTION_THRESHOLD) {
        insertionSort(lo, hi);
        return;
      }
      const mid = lo + Math.floor((hi - lo) / 2);
      mergeSort(lo, mid);
      mergeSort(mid + 1, hi);
      const left = a.slice(lo, mid + 1);
      const right = a.slice(mid + 1, hi + 1);
      let i = 0;
      let j = 0;
      let k = lo;
      pushPhase(`マージ: 位置 ${lo} … ${hi}`);
      while (i < left.length && j < right.length) {
        steps.push({
          kind: 'compare',
          lo: lo + i,
          hi: mid + 1 + j,
          arr: a.slice(),
          phase: 'merge',
        });
        if (left[i] <= right[j]) {
          a[k] = left[i];
          steps.push({
            kind: 'write',
            idx: k,
            arr: a.slice(),
            phase: 'merge',
          });
          i++;
        } else {
          a[k] = right[j];
          steps.push({
            kind: 'write',
            idx: k,
            arr: a.slice(),
            phase: 'merge',
          });
          j++;
        }
        k++;
      }
      while (i < left.length) {
        a[k] = left[i];
        steps.push({
          kind: 'write',
          idx: k,
          arr: a.slice(),
          phase: 'merge',
        });
        i++;
        k++;
      }
      while (j < right.length) {
        a[k] = right[j];
        steps.push({
          kind: 'write',
          idx: k,
          arr: a.slice(),
          phase: 'merge',
        });
        j++;
        k++;
      }
    }

    function stablePartition(lo, hi, pivot) {
      const src = a.slice(lo, hi + 1);
      const L = [];
      const R = [];
      for (let i = 0; i < src.length; i++) {
        const x = src[i];
        const abs = lo + i;
        steps.push({
          kind: 'compare',
          lo: abs,
          hi: abs,
          pivotVal: pivot,
          arr: a.slice(),
          phase: 'part',
        });
        if (x <= pivot) {
          L.push(x);
        } else {
          R.push(x);
        }
        const merged = L.concat(R);
        for (let t = 0; t < merged.length; t++) {
          a[lo + t] = merged[t];
        }
        for (let t = merged.length; t < src.length; t++) {
          a[lo + t] = src[t];
        }
        // 書き分け先だけを光らせる（左右全体を塗るとチカチカする）
        const writeIdx =
          x <= pivot ? lo + L.length - 1 : lo + L.length + R.length - 1;
        steps.push({
          kind: 'part_progress',
          writeIdx: writeIdx,
          leftLen: L.length,
          arr: a.slice(),
          pivotVal: pivot,
        });
      }
      return L.length;
    }

    function fluxRange(lo, hi) {
      const size = hi - lo + 1;
      if (size <= 1) return;
      if (size < INSERTION_THRESHOLD) {
        insertionSort(lo, hi);
        return;
      }

      const pivot = quasimedian9(a, lo, hi);
      pushPhase(`ピボット ${pivot} で位置 ${lo} … ${hi} を安定分割`);
      const left = stablePartition(lo, hi, pivot);
      const right = size - left;
      const split = lo + left;

      steps.push({
        kind: 'part_end',
        lo: lo,
        hi: hi,
        split: split,
        pivotVal: pivot,
        arr: a.slice(),
      });

      if (right === 0) {
        const src = a.slice(lo, hi + 1);
        let lt = 0;
        for (let i = 0; i < src.length; i++) {
          if (src[i] < pivot) {
            a[lo + lt] = src[i];
            lt++;
          }
        }
        let eq = lt;
        for (let i = 0; i < src.length; i++) {
          if (src[i] === pivot) {
            a[lo + eq] = src[i];
            eq++;
          }
        }
        pushPhase(
          `第二走査: 等値 ${pivot} を確定し、厳密に小さい側だけ再帰`
        );
        // 等値帯の先頭だけを印付け（帯全体は塗らない）
        if (eq > lt) {
          steps.push({
            kind: 'part_end',
            split: lo + lt,
            hi: hi,
            pivotVal: pivot,
            arr: a.slice(),
            caption: `等値帯の先頭: 位置 ${lo + lt}（値 ${pivot}）`,
          });
        }
        if (lt > 1) fluxRange(lo, lo + lt - 1);
        return;
      }

      if (left < size / UNBALANCE_RATIO || right < size / UNBALANCE_RATIO) {
        pushPhase(`不均衡のためマージソートへ切替: 位置 ${lo} … ${hi}`);
        mergeSort(lo, split - 1);
        mergeSort(split, hi);
        return;
      }

      if (left > 1) fluxRange(lo, split - 1);
      if (right > 1) fluxRange(split, hi);
    }

    if (n === 0) {
      steps.push({ kind: 'done', arr: [] });
      return steps;
    }

    pushPhase('アナライザ: 整列済み／逆順／区間の整列度を検査');
    if (isSorted(a)) {
      pushPhase('全体が昇順のため終了');
      steps.push({ kind: 'done', arr: a.slice() });
      return steps;
    }
    if (isReverseSorted(a)) {
      pushPhase('全体が降順のため反転');
      let lo = 0;
      let hi = n - 1;
      while (lo < hi) {
        const t = a[lo];
        a[lo] = a[hi];
        a[hi] = t;
        steps.push({
          kind: 'swap',
          lo: lo,
          hi: hi,
          arr: a.slice(),
          phase: 'reverse',
        });
        lo++;
        hi--;
      }
      steps.push({ kind: 'done', arr: a.slice() });
      return steps;
    }

    const q = Math.floor(n / 4);
    if (q >= 2) {
      const bounds = [0, q, q * 2, q * 3, n];
      for (let s = 0; s < 4; s++) {
        const lo = bounds[s];
        const hi = bounds[s + 1] - 1;
        if (hi - lo < 1) continue;
        const pairs = hi - lo;
        if (orderedPairs(a, lo, hi) * 2 > pairs) {
          pushPhase(`区間 ${lo} … ${hi} の整列度が高いためマージソート`);
          mergeSort(lo, hi);
        }
      }
      if (isSorted(a)) {
        steps.push({ kind: 'done', arr: a.slice() });
        return steps;
      }
    }

    fluxRange(0, n - 1);
    steps.push({ kind: 'done', arr: a.slice() });
    return steps;
  }

      function sameMultiset(a, b) {
        if (a.length !== b.length) return false;
        const sa = a.slice().sort(function (x, y) {
          return x - y;
        });

## 類似アルゴリズムとの相違点

[クワッドソート](/2026/08/17/sort-quad.html)は分割を行わず、ボトムアップのクワッドマージだけで完結する。フラックスソートはランダム寄りの入力では安定分割を主とし、整列度が高い区間・小区間・不均衡時だけクワッドソート系へ寄せる。

[ロムート分割型クイックソート](/2026/05/02/sort-quick-lumoto.html)はインプレースで不安定な分割を行う。フラックスソートは補助メモリへの二重書き込みで安定分割し、分岐の少ない走査を前提にしている。

[パターン撃退型クイックソート](/2026/07/27/sort-pattern-defeating-quick.html)も悪パターン対策のハイブリッドだが、不安定で補助メモリをほぼ使わない。フラックスソートは安定性を保ったまま分割とマージを行き来する。

[ティムソート](/2026/05/23/sort-tim.html)は自然ランの検出とギャロッピング併合が中心である。フラックスソートのアナライザは先頭で 4 区間の整列度を見る程度に留め、以降は分割側の適応に寄せる。
