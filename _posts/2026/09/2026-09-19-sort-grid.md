## グリッドソートを使用する

グリッドソート (`gridsort`) は、要素を 2 軸の自己平衡配列（バイナリグリッド）へ挿入し、フロア順に読み出すことで整列する安定・適応的な分割型比較ソートである。[キューブソート](/2026/09/18/sort-cube.html)のバイナリキューブを 1 段簡略化した姉妹アルゴリズムで、ルックアップテーブル上の二分探索だけで挿入先バケットを決める。

本稿の実装は説明用に簡略化し、本番実装で使われるバウンドレス／適応的二分探索やクワッドソートによるバケット整列の代わりに、通常の二分探索と挿入ソートを用いる。バケット容量はおよそ $$\sqrt{n}$$ にとり、約 $$2\sqrt{n}$$ 本のバケットがそれぞれ $$\sqrt{n}$$ 要素を収める想定に揃える。

1.  **容量の決定**: 要素数 $$n$$ の平方根に近いバケット容量をとり、フロア列とバケット本数が同程度のスケールになるよう保つ。
2.  **挿入**: キーに対し各バケットのフロア（先頭値）を二分探索し、該当バケット末尾へ追加する。フロアは分割直後の最小値で、未ソートの追加では更新しない。
3.  **オーバーフロー分割**: バケットが容量に達したら全体をソートし、半分ずつに分けて隣へ新しいバケットを挿入する。両側のフロアをルックアップへ反映する。
4.  **仕上げ**: 残った未ソートバケットをソートし、フロア昇順に連結して元配列へ書き戻す。

```pseudocode
procedure grid_sort(A)
  n = length(A)
  if n < 2 then return
  capacity = round_up_pow2(sqrt(n))   // 下限あり
  buckets = [bucket with floor A[0], items [A[0]]]
  for i = 1 .. n-1
    key = A[i]
    b = binary_search_floor(buckets, key)
    append key to buckets[b]
    if length(buckets[b]) >= capacity then
      sort(buckets[b]); split into two halves; insert right half after b
  for each bucket B in floor order
    sort(B) if needed
    append B to output
  copy output back to A
```

整列済みや逆順では、ほぼ常に端のバケットへ追記されて分割が起きにくく、比較はおおよそ $$O(n)$$ に近づく。平均・最悪は各挿入が $$O(\log n)$$ 相当のため $$O(n \log n)$$、追加メモリはグリッド本体で $$O(n)$$ である。バケット内を安定ソートし末尾追加するため、全体としても安定である。

デモでは視認性のため容量を 4 に固定し、上段に未挿入または出力の枠、その下にフロア付き Y バケットを示す。開始時は全要素が未挿入にあり、先頭キーから最初のバケットを作る。

<script>
window.DemoSort && DemoSort.boot('grid-sort-demo', function (root) {
  const CAPACITY = 4;
  const barClass = 'sort-demo__bar';
  const initialValues = [7, 2, 9, 4, 1, 8, 3, 6, 5, 0, 11, 10];
  const initialCaption =
    'シャッフルまたはステップでグリッドソートを開始';

  function prefersReducedMotion() {
    try {
      return window.matchMedia('(prefers-reduced-motion: reduce)').matches;
    } catch (_e) {
      return false;
    }
  }

  function nextFrame() {
    return new Promise(function (resolve) {
      requestAnimationFrame(function () {
        requestAnimationFrame(resolve);
      });
    });
  }

  function cloneBuckets(bs) {
    return bs.map(function (b) {
      return {
        floor: b.floor,
        items: b.items.slice(),
        sorted: b.sorted,
      };
    });
  }

  function upperBoundFloor(floors, key) {
    let lo = 0;
    let hi = floors.length;
    while (lo < hi) {
      const mid = (lo + hi) >> 1;
      if (floors[mid] <= key) lo = mid + 1;
      else hi = mid;
    }
    return Math.max(0, lo - 1);
  }

  function ensureSorted(bucket) {
    if (bucket.sorted) return;
    bucket.items.sort(function (a, b) {
      return a - b;
    });
    bucket.sorted = true;
    if (bucket.items.length) bucket.floor = bucket.items[0];
  }

  function splitBucket(buckets, bi) {
    const bucket = buckets[bi];
    ensureSorted(bucket);
    const mid = (bucket.items.length / 2) | 0;
    if (mid <= 0 || mid >= bucket.items.length) return;
    const rightItems = bucket.items.slice(mid);
    bucket.items = bucket.items.slice(0, mid);
    bucket.floor = bucket.items[0];
    bucket.sorted = true;
    buckets.splice(bi + 1, 0, {
      floor: rightItems[0],
      items: rightItems,
      sorted: true,
    });
  }

  function generateSteps(initial) {
    const steps = [];
    const a = initial.slice();
    const n = a.length;
    if (n < 2) {
      steps.push({
        kind: 'done',
        incoming: [],
        buckets: [],
        out: a.slice(),
        active: null,
      });
      return steps;
    }

    const buckets = [];
    const incoming = a.slice();

    function push(kind, extra) {
      steps.push(
        Object.assign(
          {
            kind: kind,
            incoming: incoming.slice(),
            buckets: cloneBuckets(buckets),
            out: null,
            active: null,
          },
          extra || {}
        )
      );
    }

    push('start');

    while (incoming.length) {
      const key = incoming[0];

      if (!buckets.length) {
        push('locate', { key: key, active: null });
        incoming.shift();
        buckets.push({ floor: key, items: [key], sorted: true });
        push('insert', { key: key, active: 0 });
        continue;
      }

      const floors = buckets.map(function (b) {
        return b.floor;
      });
      const bi = upperBoundFloor(floors, key);
      push('locate', { key: key, active: bi });

      incoming.shift();
      buckets[bi].items.push(key);
      buckets[bi].sorted = false;
      push('insert', { key: key, active: bi });

      if (buckets[bi].items.length >= CAPACITY) {
        push('overflow', { key: key, active: bi });
        ensureSorted(buckets[bi]);
        push('sort_bucket', { active: bi });
        splitBucket(buckets, bi);
        push('split', { active: bi });
      }
    }

    const out = [];
    for (let bi = 0; bi < buckets.length; bi++) {
      const bucket = buckets[bi];
      if (!bucket.items.length) continue;
      if (!bucket.sorted) {
        ensureSorted(bucket);
        push('sort_bucket', {
          active: bi,
          out: out.slice(),
        });
      }
      const moved = bucket.items.slice();
      push('finish_bucket', {
        active: bi,
        out: out.slice(),
      });
      out.push.apply(out, moved);
      bucket.items = [];
      push('emit_bucket', {
        active: bi,
        out: out.slice(),
        movedCount: moved.length,
      });
    }
    steps.push({
      kind: 'done',
      incoming: [],
      buckets: [],
      out: out.slice(),
      active: null,
    });
    return steps;
  }

  async function flipToStep(barsEl, s, values) {
    if (prefersReducedMotion()) {
      mountGrid(barsEl, s, values);
      return;
    }
    const fromRects = collectBarRects(barsEl);
    mountGrid(barsEl, s, values);
    const bars = Array.prototype.slice.call(
      barsEl.querySelectorAll('.sort-demo__bar')
    );
    const moving = [];
    for (let i = 0; i < bars.length; i++) {
      const el = bars[i];
      const from = fromRects.get(el.getAttribute('title'));
      if (!from) continue;
      const to = el.getBoundingClientRect();
      const dx = from.left - to.left;
      const dy = from.top - to.top;
      if (Math.abs(dx) < 0.5 && Math.abs(dy) < 0.5) continue;
      moving.push({ el: el, dx: dx, dy: dy });
    }
    if (!moving.length) return;
    for (let i = 0; i < moving.length; i++) {
      const m = moving[i];
      m.el.style.transition = 'none';
      m.el.style.transform =
        'translate(' + m.dx + 'px,' + m.dy + 'px)';
      m.el.style.willChange = 'transform';
    }
    void barsEl.offsetWidth;
    await nextFrame();
    await Promise.all(
      moving.map(function (m) {
        m.el.style.transition = 'transform 360ms ease-in-out';
        m.el.style.transform = '';
        return DemoSort.transitionPromise(m.el);
      })
    );
    for (let i = 0; i < moving.length; i++) {
      const el = moving[i].el;
      el.style.transition = '';
      el.style.transform = '';
      el.style.willChange = '';
    }
  }

  function collectBarRects(container) {
    const map = new Map();
    const bars = container.querySelectorAll('.sort-demo__bar');
    for (let i = 0; i < bars.length; i++) {
      const el = bars[i];
      map.set(el.getAttribute('title'), el.getBoundingClientRect());
    }
    return map;
  }

  function barHeight(val, mn, mx) {
    const span = Math.max(mx - mn, 1);
    return 22 + ((val - mn) / span) * 70;
  }

  function mountGrid(container, s, values) {
    container.innerHTML = '';
    container.setAttribute('role', 'group');
    container.setAttribute(
      'aria-label',
      'グリッドソートの現在の状態（未挿入または出力・バイナリグリッド）。'
    );
    const wrap = document.createElement('div');
    wrap.className = 'sort-demo-grid-wrap';
    const mn = Math.min.apply(null, values);
    const mx = Math.max.apply(null, values);

    function mkBar(val, role) {
      const bar = document.createElement('div');
      bar.className = barClass;
      bar.style.height = barHeight(val, mn, mx) + 'px';
      bar.setAttribute('title', String(val));
      if (role) bar.setAttribute('data-role', role);
      bar.setAttribute(
        'aria-label',
        DemoSort.barAccessibilityLabelSimple(String(val), role)
      );
      return bar;
    }

    const showOut = !!(s.out && s.out.length);
    if (showOut) {
      const done = document.createElement('div');
      done.className = 'sort-demo-grid__done';
      for (let i = 0; i < s.out.length; i++) {
        let role = null;
        if (
          s.kind === 'emit_bucket' &&
          i >= s.out.length - (s.movedCount || 0)
        ) {
          role = 'write';
        }
        done.appendChild(mkBar(s.out[i], role));
      }
      wrap.appendChild(done);
    } else {
      const inc = document.createElement('div');
      inc.className = 'sort-demo-grid__incoming';
      if (s.incoming && s.incoming.length) {
        for (let i = 0; i < s.incoming.length; i++) {
          const role = i === 0 && s.kind === 'locate' ? 'key' : null;
          inc.appendChild(mkBar(s.incoming[i], role));
        }
      }
      wrap.appendChild(inc);
    }

    if (s.buckets && s.buckets.length && s.kind !== 'done') {
      const row = document.createElement('div');
      row.className = 'sort-demo-grid__row';
      for (let bi = 0; bi < s.buckets.length; bi++) {
        const b = s.buckets[bi];
        const yEl = document.createElement('div');
        yEl.className = 'sort-demo-grid__y';
        if (s.active === bi) {
          yEl.classList.add('sort-demo-grid__y--active');
        }
        const yLab = document.createElement('div');
        yLab.className = 'sort-demo-grid__y-label';
        yLab.textContent =
          'Y floor ' + b.floor + (b.sorted ? '' : ' ·未整列');
        yEl.appendChild(yLab);
        const bars = document.createElement('div');
        bars.className = 'sort-demo-grid__y-bars';
        for (let k = 0; k < b.items.length; k++) {
          let role = null;
          if (
            s.active === bi &&
            (s.kind === 'insert' || s.kind === 'overflow') &&
            k === b.items.length - 1
          ) {
            role = 'insert';
          }
          bars.appendChild(mkBar(b.items[k], role));
        }
        yEl.appendChild(bars);
        row.appendChild(yEl);
      }
      wrap.appendChild(row);
    }

    container.appendChild(wrap);
  }

  DemoSort.attachPlayback({
    root: root,
    dataAttr: 'data-grid',
    barClass: barClass,
    initialValues: initialValues,
    initialCaption: initialCaption,
    generateSteps: generateSteps,
    rebuild: function (api, v) {
      api.values = v;
      api.steps = generateSteps(v.slice());
      api.idx = 0;
      const first = api.steps[0];
      mountGrid(api.barsEl, first, v);
      api.setCaption(initialCaption);
    },
    applyStep: async function (api, s) {
      const barsEl = api.barsEl;
      if (
        s.kind === 'insert' ||
        s.kind === 'sort_bucket' ||
        s.kind === 'split' ||
        s.kind === 'emit_bucket'
      ) {
        await flipToStep(barsEl, s, api.values);
      } else {
        mountGrid(barsEl, s, api.values);
      }

      if (s.kind === 'start') {
        api.setCaption('未挿入の全要素から開始（まだバケットはない）');
      } else if (s.kind === 'locate') {
        if (s.active == null) {
          api.setCaption(
            '先頭キー ' + s.key + ' で最初の Y バケットを作る'
          );
        } else {
          api.setCaption(
            'キー ' +
              s.key +
              ' の挿入先をフロア二分探索で特定（強調バケット）'
          );
        }
      } else if (s.kind === 'insert') {
        api.setCaption('キー ' + s.key + ' を Y バケット末尾へ追加');
      } else if (s.kind === 'overflow') {
        api.setCaption(
          'Y バケットが容量 ' + CAPACITY + ' に達したので分割準備'
        );
      } else if (s.kind === 'sort_bucket') {
        api.setCaption('Y バケット内をソートする');
      } else if (s.kind === 'split') {
        api.setCaption('ソート済み Y を半分に分割し、新しいバケットを挿入');
      } else if (s.kind === 'finish_bucket') {
        api.setCaption('整列済みバケットを出力へつなぐ準備');
      } else if (s.kind === 'emit_bucket') {
        api.setCaption('整列済みバケットを出力へ移す');
      } else if (s.kind === 'done') {
        api.setCaption('ソート完了（フロア順に連結）');
      }
    },
    stepPauseMs: 300,
  });
});
</script>

未挿入の先頭が次のキーで、枠付きの Y が現在の挿入先である。未挿入と出力は同じ高さの枠を切り替えて表示する。分割後は各バケットのフロア（先頭値）だけが次の二分探索に使われる。

## 類似アルゴリズムとの相違点

[キューブソート](/2026/09/18/sort-cube.html)は同じ分割挿入を X・Y・Z の 3 軸で行う上位概念である。グリッドはルックアップを 1 段に落とすため実装が単純で、キューブの分割挿入を理解する足場になる。

[バケットソート](/2026/06/23/sort-bucket.html)は値域やハッシュで桶を決めるのに対し、グリッドソートは比較だけでフロア列を二分探索する。分布が一様でなくても最悪 $$O(n \log n)$$ を保てる。

[クワッドソート](/2026/08/17/sort-quad.html)は本番グリッドソートがバケット整列に用いる部品である。本稿では挿入ソートに置き換えている。

[フラックスソート](/2026/08/18/sort-flux.html)はトップダウン分割で同系統の安定適応ソートを実現する。グリッドはボトムアップにバケットを育てる点が異なる。
