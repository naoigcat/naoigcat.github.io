## キューブソートを使用する

キューブソート (`cubesort`) は、要素を自己平衡な多次元配列（バイナリキューブ）へ挿入し、軸に沿って読み出すことで整列する安定・適応的な分割型比較ソートである。

X・Y・Z の 3 軸が同程度の長さに保たれると構造が立方体に近いためこの名がある。各挿入では軸ごとに二分探索してバケットを決め、溢れた Z バケットだけをまとめてソートして二分する。全要素を入れ終わったら未ソートのバケットを仕上げ、軸順に連結すれば完了する。本稿の実装は説明用に簡略化し、本番実装で使われるモノバウンド二分探索やクワッドソートによるバケット整列の代わりに、通常の二分探索と挿入ソートを用いる。

1.  **容量の決定**: 要素数 $$n$$ の立方根に近いバケット容量をとり、3 軸が同程度の長さになるよう保つ。
2.  **挿入**: キーに対し X 軸（Y 行のフロア）・Y 軸（Z バケットのフロア）を二分探索し、該当 Z バケット末尾へ追加する。フロアはバケット先頭（分割直後の最小値）で、未ソートの追加では更新しない。
3.  **オーバーフロー分割**: Z が容量に達したらバケット全体をソートし、半分ずつに分けて隣へ挿入する。Y 行の Z 本数が容量の約 2 倍を超えたら Y 行自体も半分に分割して X 軸へ差し込む。
4.  **仕上げ**: 残った未ソートバケットをソートし、X → Y → Z の順に連結して元配列へ書き戻す。

```pseudocode
procedure cube_sort(A)
  n = length(A)
  if n < 2 then return
  capacity = round_up_pow2(cbrt(n))   // 下限あり
  cube = X axis with one Y row holding one Z bucket [A[0]]
  for i = 1 .. n-1
    key = A[i]
    y = binary_search_floor(cube.X, key)
    z = binary_search_floor(y.Z, key)
    append key to z
    if length(z) >= capacity then
      sort(z); split z into two halves; insert right half after z
      if length(y.Z) > 2 * capacity then
        split y into two Y rows on the X axis
  for each Z bucket B in cube order
    sort(B) if needed
    append B to output
  copy output back to A
```

整列済みや逆順では、ほぼ常に端のバケットへ追記されて分割が起きにくく、比較はおおよそ $$O(n)$$ に近づく。平均・最悪は各挿入が $$O(\log n)$$ 相当のため $$O(n \log n)$$、追加メモリはキューブ本体で $$O(n)$$ である。バケット内を安定ソートし末尾追加するため、全体としても安定である。

デモでは視認性のため容量を 4 に固定し、上段に未挿入列、下に X 行ごとの Z バケットを示す。

<script>
window.DemoSort && DemoSort.boot('cube-sort-demo', function (root) {
  const CAPACITY = 4;
  const barClass = 'sort-demo__bar';
  const initialValues = [7, 2, 9, 4, 1, 8, 3, 6, 5, 0, 11, 10];
  const initialCaption =
    'シャッフルまたはステップでキューブソートを開始';

  function cloneCube(xs) {
    return xs.map(function (y) {
      return y.map(function (z) {
        return { floor: z.floor, items: z.items.slice(), sorted: z.sorted };
      });
    });
  }

  function upperBoundFloor(floors, key) {
    let lo = 0;
    let hi = floors.length;
    while (lo < hi) {
      const mid = (lo + hi) >> 1;
      if (floors[mid] <= key) lo = mid + 1;
      else hi = mid;
    }
    return Math.max(0, lo - 1);
  }

  function ensureSorted(bucket) {
    if (bucket.sorted) return;
    bucket.items.sort(function (a, b) {
      return a - b;
    });
    bucket.sorted = true;
    if (bucket.items.length) bucket.floor = bucket.items[0];
  }

  function splitZ(y, zi) {
    const bucket = y[zi];
    ensureSorted(bucket);
    const mid = (bucket.items.length / 2) | 0;
    if (mid <= 0 || mid >= bucket.items.length) return false;
    const rightItems = bucket.items.slice(mid);
    bucket.items = bucket.items.slice(0, mid);
    bucket.floor = bucket.items[0];
    bucket.sorted = true;
    y.splice(zi + 1, 0, {
      floor: rightItems[0],
      items: rightItems,
      sorted: true,
    });
    return y.length > CAPACITY * 2;
  }

  function splitY(xs, yi) {
    const y = xs[yi];
    const mid = (y.length / 2) | 0;
    if (mid <= 0 || mid >= y.length) return;
    const right = y.splice(mid);
    xs.splice(yi + 1, 0, right);
  }

  function generateSteps(initial) {
    const steps = [];
    const a = initial.slice();
    const n = a.length;
    if (n < 2) {
      steps.push({
        kind: 'done',
        incoming: [],
        xs: [],
        out: a.slice(),
        active: null,
      });
      return steps;
    }

    const xs = [[{ floor: a[0], items: [a[0]], sorted: true }]];
    const incoming = a.slice(1);

    function push(kind, extra) {
      steps.push(
        Object.assign(
          {
            kind: kind,
            incoming: incoming.slice(),
            xs: cloneCube(xs),
            out: null,
            active: null,
          },
          extra || {}
        )
      );
    }

    push('start', { active: { yi: 0, zi: 0 } });

    while (incoming.length) {
      const key = incoming[0];
      const yFloors = xs.map(function (y) {
        return y[0].floor;
      });
      const yi = upperBoundFloor(yFloors, key);
      const zFloors = xs[yi].map(function (z) {
        return z.floor;
      });
      const zi = upperBoundFloor(zFloors, key);
      push('locate', { key: key, active: { yi: yi, zi: zi } });

      incoming.shift();
      xs[yi][zi].items.push(key);
      xs[yi][zi].sorted = false;
      push('insert', { key: key, active: { yi: yi, zi: zi } });

      if (xs[yi][zi].items.length >= CAPACITY) {
        push('overflow', { key: key, active: { yi: yi, zi: zi } });
        const needSplitY = splitZ(xs[yi], zi);
        push('split_z', { active: { yi: yi, zi: zi } });
        if (needSplitY) {
          splitY(xs, yi);
          push('split_y', { active: { yi: yi, zi: 0 } });
        }
      }
    }

    const out = [];
    for (let yi = 0; yi < xs.length; yi++) {
      for (let zi = 0; zi < xs[yi].length; zi++) {
        ensureSorted(xs[yi][zi]);
        push('finish_bucket', { active: { yi: yi, zi: zi } });
        out.push.apply(out, xs[yi][zi].items);
      }
    }
    steps.push({
      kind: 'done',
      incoming: [],
      xs: cloneCube(xs),
      out: out.slice(),
      active: null,
    });
    return steps;
  }

  function barHeight(val, mn, mx) {
    const span = Math.max(mx - mn, 1);
    return 22 + ((val - mn) / span) * 70;
  }

  function mountCube(container, s, values) {
    container.innerHTML = '';
    container.setAttribute('role', 'group');
    container.setAttribute(
      'aria-label',
      'キューブソートの現在の状態（未挿入・バイナリキューブ・出力）。'
    );
    const wrap = document.createElement('div');
    wrap.className = 'sort-demo-cube-wrap';
    const mn = Math.min.apply(null, values);
    const mx = Math.max.apply(null, values);

    function mkBar(val, role) {
      const bar = document.createElement('div');
      bar.className = barClass;
      bar.style.height = barHeight(val, mn, mx) + 'px';
      bar.setAttribute('title', String(val));
      if (role) bar.setAttribute('data-role', role);
      bar.setAttribute(
        'aria-label',
        DemoSort.barAccessibilityLabelSimple(String(val), role)
      );
      return bar;
    }

    if (s.incoming && s.incoming.length) {
      const inc = document.createElement('div');
      inc.className = 'sort-demo-cube__incoming';
      for (let i = 0; i < s.incoming.length; i++) {
        const role = i === 0 && s.kind === 'locate' ? 'key' : null;
        inc.appendChild(mkBar(s.incoming[i], role));
      }
      wrap.appendChild(inc);
    }

    if (s.out) {
      const done = document.createElement('div');
      done.className = 'sort-demo-cube__done';
      for (let i = 0; i < s.out.length; i++) {
        done.appendChild(mkBar(s.out[i], 'sorted'));
      }
      wrap.appendChild(done);
    } else if (s.xs) {
      for (let yi = 0; yi < s.xs.length; yi++) {
        const xEl = document.createElement('div');
        xEl.className = 'sort-demo-cube__x';
        const lab = document.createElement('div');
        lab.className = 'sort-demo-cube__x-label';
        lab.textContent = 'X[' + yi + ']（Y 行）';
        xEl.appendChild(lab);
        const row = document.createElement('div');
        row.className = 'sort-demo-cube__y-row';
        for (let zi = 0; zi < s.xs[yi].length; zi++) {
          const z = s.xs[yi][zi];
          const zEl = document.createElement('div');
          zEl.className = 'sort-demo-cube__z';
          if (s.active && s.active.yi === yi && s.active.zi === zi) {
            zEl.classList.add('sort-demo-cube__z--active');
          }
          const zLab = document.createElement('div');
          zLab.className = 'sort-demo-cube__z-label';
          zLab.textContent =
            'Z floor ' + z.floor + (z.sorted ? '' : ' ·未整列');
          zEl.appendChild(zLab);
          const bars = document.createElement('div');
          bars.className = 'sort-demo-cube__z-bars';
          for (let k = 0; k < z.items.length; k++) {
            let role = null;
            if (
              s.active &&
              s.active.yi === yi &&
              s.active.zi === zi &&
              (s.kind === 'insert' || s.kind === 'overflow') &&
              k === z.items.length - 1
            ) {
              role = 'insert';
            } else if (z.sorted) {
              role = 'sorted';
            } else {
              role = 'range';
            }
            bars.appendChild(mkBar(z.items[k], role));
          }
          zEl.appendChild(bars);
          row.appendChild(zEl);
        }
        xEl.appendChild(row);
        wrap.appendChild(xEl);
      }
    }

    container.appendChild(wrap);
  }

  DemoSort.attachPlayback({
    root: root,
    dataAttr: 'data-cube',
    barClass: barClass,
    initialValues: initialValues,
    initialCaption: initialCaption,
    generateSteps: generateSteps,
    rebuild: function (api, v) {
      api.values = v;
      api.steps = generateSteps(v.slice());
      api.idx = 0;
      const first = api.steps[0];
      mountCube(api.barsEl, first, v);
      api.setCaption(initialCaption);
    },
    applyStep: async function (api, s) {
      mountCube(api.barsEl, s, api.values);
      if (s.kind === 'start') {
        api.setCaption('先頭要素で最初の Z バケットを作る');
      } else if (s.kind === 'locate') {
        api.setCaption(
          'キー ' +
            s.key +
            ' の挿入先を X・Y のフロア二分探索で特定（強調バケット）'
        );
      } else if (s.kind === 'insert') {
        api.setCaption('キー ' + s.key + ' を Z バケット末尾へ追加');
      } else if (s.kind === 'overflow') {
        api.setCaption(
          'Z バケットが容量 ' + CAPACITY + ' に達したので分割準備'
        );
      } else if (s.kind === 'split_z') {
        api.setCaption('Z をソートして半分に分割し、新しいバケットを挿入');
      } else if (s.kind === 'split_y') {
        api.setCaption('Y 行が太ったので X 軸上で Y 行を二分');
      } else if (s.kind === 'finish_bucket') {
        api.setCaption('残バケットをソートして出力へつなぐ準備');
      } else if (s.kind === 'done') {
        api.setCaption('ソート完了（軸順に連結）');
      }
    },
    stepPauseMs: 240,
  });
});
</script>

未挿入の先頭が次のキーで、枠付きの Z が現在の挿入先である。分割後は各バケットのフロア（先頭値）だけが次の二分探索に使われる。

## 類似アルゴリズムとの相違点

[グリッドソート](https://github.com/scandum/gridsort)はバイナリキューブを 2 軸に簡略化した姉妹アルゴリズムで、キューブソートの分割挿入を理解する足場になる。本稿の 3 軸版はその上位概念に近い。

[バケットソート](/2026/06/23/sort-bucket.html)は値域やハッシュで桶を決めるのに対し、キューブソートは比較だけでフロア列を二分探索する。分布が一様でなくても最悪 $$O(n \log n)$$ を保てる。

[二分木ソート](/2026/05/12/sort-tree.html)も挿入構造から読み出す点は似るが、節点がポインタ木になる。キューブは配列状の軸に乗るため走査が単純で、キャッシュ局所性を取りやすい。

[クワッドソート](/2026/08/17/sort-quad.html)は本番キューブソートがバケット整列に用いる部品である。本稿では挿入ソートに置き換えている。
