## オクトソートを使用する

オクトソート (`octosort`) は、[ウィキソート](/2026/05/31/sort-wiki.html)のブロックマージ骨格に[クワッドソート](/2026/08/17/sort-quad.html)由来の部品を組み込んだ、安定・適応的なインプレース整列である。

本記事の実装は説明用に簡略化し、大きいレベルでは本番の内部バッファ＋ブロックタグ付けではなく、固定キャッシュと回転ベースのマージで代用する。

1.  **オクトスワップ**: 長さ 4〜8 の小区間を整える。先頭 4 要素が厳密な降順パターンなら逆順ランの延長を試み、そうでなければソートネットワークと末尾挿入で昇順にする。連続する逆順ランは後で一括反転する。
2.  **逆順ランの反転**: オクトスワップが残した逆順区間を反転する。配列全体が逆順ならここで終了する。
3.  **クワッドマージ**: 4 本の整列済みブロックが固定キャッシュに収まるレベルでは、2 組ずつキャッシュへマージし、続けて本配列へ戻す（ピンポン）。境界がすべて昇順ならスキップする。逆順検出はオクトスワップ側に任せる。
4.  **テイルマージ**: 2 本だけマージするレベルでは、片側がキャッシュに収まるなら退避してマージする。
5.  **インプレースマージ**: どちらもキャッシュに収まらないときは、モノバウンド二分探索で分割点を決め、グリースミルズ回転（ブロックスワップ）で区間を組み替えて再帰する。

```pseudocode

  // 進行中の逆順ランがあれば先に反転してから整列

procedure rotate_gries_mills(A, amount, range, cache)
  // 片側が cache に収まるなら退避移動、そうでなければブロックスワップ回転

procedure octosort(A)
  rev := nil
  for each WikiIterator run R of size 4..8
    rev := octo_swap(A, R, rev)
  if rev ≠ nil then
    reverse(A[rev .. end)); if rev = 0 then return
  level := 4
  while level < length(A)
    if 4 * level fits in fixed_cache then
      for each aligned span of 4 runs
        quad_merge_via_cache(span)   // 境界昇順なら skip
    else
      for each adjacent pair (L, R)
        if L or R fits in cache then
          external_or_tail_merge(L, R)
        else
          monobound_rotate_merge(L, R)
    level := next_merge_level(level)
```

最良は整列済み・全逆順の検出により $$O(n)$$、平均・最悪は $$O(n \log n)$$ である。既定では数百要素程度の固定スタックキャッシュを使い、補助記憶は $$O(1)$$（設定次第で $$n/2$$ まで広げられる）の安定ソートである。

デモでは視認性のためラン幅を 4 に揃え、大きい区間のマージもキャッシュ経由の二分マージで示す。

<script>
window.DemoSort && DemoSort.boot('octo-sort-demo', function (root) {
  const RUN = 4;

  function buildDisplay(a, lo, tmp) {
    const d = a.slice();
    for (let t = 0; t < tmp.length; t++) {
      d[lo + t] = tmp[t];
    }
    return d;
  }

  function rangePairs(lo, hiInclusive, role) {
    const pairs = [];
    for (let k = lo; k <= hiInclusive; k++) {
      pairs.push([k, role]);
    }
    return pairs;
  }

  function pushSwap(a, i, j, steps, phase) {
    if (i === j) return;
    const t = a[i];
    a[i] = a[j];
    a[j] = t;
    steps.push({ kind: 'swap', lo: i, hi: j, phase: phase, arr: a.slice() });
  }

  function isReverse4(a, i) {
    return a[i] > a[i + 1] && a[i + 2] > a[i + 3] && a[i + 1] > a[i + 2];
  }

  function rangeNonIncreasing(a, lo, hi) {
    for (let i = lo + 1; i < hi; i++) {
      if (a[i - 1] < a[i]) return false;
    }
    return true;
  }

  function swap4Network(a, i0, steps) {
    const i1 = i0 + 1;
    const i2 = i0 + 2;
    const i3 = i0 + 3;
    if (a[i0] > a[i1]) pushSwap(a, i0, i1, steps, 'octo');
    if (a[i2] > a[i3]) pushSwap(a, i2, i3, steps, 'octo');
    if (a[i0] > a[i2]) pushSwap(a, i0, i2, steps, 'octo');
    if (a[i1] > a[i3]) pushSwap(a, i1, i3, steps, 'octo');
    if (a[i1] > a[i2]) pushSwap(a, i1, i2, steps, 'octo');
  }

  function reverseRange(a, lo, hi, steps) {
    steps.push({ kind: 'reverse', lo: lo, hi: hi - 1, arr: a.slice() });
    let i = lo;
    let j = hi - 1;
    while (i < j) {
      pushSwap(a, i, j, steps, 'reverse');
      i++;
      j--;
    }
  }

  function mergeTwo(a, lo, mid, hi, steps) {
    steps.push({ kind: 'merge_start', lo: lo, mid: mid, hi: hi - 1, arr: a.slice() });
    const tmp = [];
    let i = lo;
    let j = mid;
    while (i < mid && j < hi) {
      steps.push({
        kind: 'merge_compare',
        i: i,
        j: j,
        arr: buildDisplay(a, lo, tmp),
      });
      if (a[i] <= a[j]) {
        tmp.push(a[i]);
        i++;
      } else {
        tmp.push(a[j]);
        j++;
      }
      steps.push({
        kind: 'merge_write',
        writePos: lo + tmp.length - 1,
        arr: buildDisplay(a, lo, tmp),
      });
    }
    while (i < mid) {
      tmp.push(a[i]);
      i++;
      steps.push({
        kind: 'merge_write',
        writePos: lo + tmp.length - 1,
        arr: buildDisplay(a, lo, tmp),
      });
    }
    while (j < hi) {
      tmp.push(a[j]);
      j++;
      steps.push({
        kind: 'merge_write',
        writePos: lo + tmp.length - 1,
        arr: buildDisplay(a, lo, tmp),
      });
    }
    for (let t = 0; t < tmp.length; t++) {
      a[lo + t] = tmp[t];
    }
    steps.push({ kind: 'merge_done', lo: lo, hi: hi - 1, arr: a.slice() });
  }

  function generateSteps(initial) {
    const a = initial.slice();
    const steps = [];
    const n = a.length;
    if (n === 0) {
      steps.push({ kind: 'done', arr: a.slice() });
      return steps;
    }

    let revStart = -1;
    for (let start = 0; start < n; start += RUN) {
      const end = Math.min(start + RUN, n);
      const len = end - start;
      steps.push({ kind: 'octo_range', lo: start, hi: end - 1, arr: a.slice() });

      if (len < 4) {
        if (revStart >= 0) {
          reverseRange(a, revStart, start, steps);
          revStart = -1;
        }
        for (let i = start + 1; i < end; i++) {
          let j = i;
          while (j > start && a[j - 1] > a[j]) {
            pushSwap(a, j - 1, j, steps, 'octo');
            j--;
          }
        }
        continue;
      }

      const canExtend =
        revStart < 0 || (start > 0 && a[start - 1] >= a[start]);
      if (isReverse4(a, start) && canExtend && rangeNonIncreasing(a, start, end)) {
        if (revStart < 0) revStart = start;
        steps.push({ kind: 'rev_extend', lo: revStart, hi: end - 1, arr: a.slice() });
        continue;
      }

      if (revStart >= 0) {
        reverseRange(a, revStart, start, steps);
        revStart = -1;
      }

      if (isReverse4(a, start)) {
        pushSwap(a, start, start + 3, steps, 'octo');
        pushSwap(a, start + 1, start + 2, steps, 'octo');
      } else {
        swap4Network(a, start, steps);
      }
      steps.push({ kind: 'octo_done', lo: start, hi: end - 1, arr: a.slice() });
    }

    if (revStart >= 0) {
      reverseRange(a, revStart, n, steps);
      if (revStart === 0) {
        steps.push({ kind: 'done', arr: a.slice() });
        return steps;
      }
    }

    let width = RUN;
    while (width < n) {
      steps.push({ kind: 'level_start', width: width, arr: a.slice() });
      for (let lo = 0; lo + width < n; lo += width * 2) {
        const mid = lo + width;
        const hi = Math.min(lo + width * 2, n);
        if (a[mid - 1] <= a[mid]) {
          steps.push({ kind: 'skip', lo: lo, hi: hi - 1, arr: a.slice() });
          continue;
        }
        mergeTwo(a, lo, mid, hi, steps);
      }
      steps.push({ kind: 'level_done', width: width, arr: a.slice() });
      width *= 2;
    }

    steps.push({ kind: 'done', arr: a.slice() });
    return steps;
  }

  DemoSort.attachPlayback({
    root: root,
    dataAttr: 'data-octo',
    initialValues: [8, 7, 6, 5, 1, 3, 4, 2, 12, 11, 10, 9, 13, 15, 14],
    initialCaption:
      'オクトソートのデモ（オクト範囲は青、比較はオレンジ、書き込みは緑、交換は紫）',
    barClass: 'sort-demo__bar',
    generateSteps: generateSteps,
    applyStep: async function (api, s) {
      const barsEl = api.barsEl;
      if (s.kind === 'octo_range') {
        api.mountBars(barsEl, s.arr);
        DemoSort.assignRoles(barsEl, rangePairs(s.lo, s.hi, 'range'));
        api.setCaption('オクトスワップ: 区間 ' + s.lo + ' … ' + s.hi);
        return;
      }
      if (s.kind === 'rev_extend') {
        api.mountBars(barsEl, s.arr);
        DemoSort.assignRoles(barsEl, rangePairs(s.lo, s.hi, 'range'));
        api.setCaption('逆順ランを延長: ' + s.lo + ' … ' + s.hi);
        return;
      }
      if (s.kind === 'octo_done') {
        api.mountBars(barsEl, s.arr);
        DemoSort.assignRoles(barsEl, rangePairs(s.lo, s.hi, 'sorted'));
        api.setCaption('小区間の整列完了: ' + s.lo + ' … ' + s.hi);
        return;
      }
      if (s.kind === 'reverse') {
        api.mountBars(barsEl, s.arr);
        DemoSort.assignRoles(barsEl, rangePairs(s.lo, s.hi, 'range'));
        api.setCaption('逆順ランを反転: ' + s.lo + ' … ' + s.hi);
        return;
      }
      if (s.kind === 'swap') {
        DemoSort.assignRoles(barsEl, [
          [s.lo, 'swap'],
          [s.hi, 'swap'],
        ]);
        api.setCaption(
          s.phase === 'reverse' ? '反転の交換' : 'オクトスワップの交換'
        );
        await DemoSort.flipSwap(barsEl, s.lo, s.hi);
        DemoSort.clearRoles(barsEl);
        return;
      }
      if (s.kind === 'level_start') {
        api.mountBars(barsEl, s.arr);
        DemoSort.clearRoles(barsEl);
        api.setCaption(
          'ボトムアップマージ: ラン幅 ' + s.width + ' → ' + (s.width * 2)
        );
        return;
      }
      if (s.kind === 'skip') {
        api.mountBars(barsEl, s.arr);
        DemoSort.assignRoles(barsEl, rangePairs(s.lo, s.hi, 'sorted'));
        api.setCaption('境界昇順のためマージを省略: ' + s.lo + ' … ' + s.hi);
        return;
      }
      if (s.kind === 'merge_start') {
        api.mountBars(barsEl, s.arr);
        DemoSort.assignRoles(barsEl, rangePairs(s.lo, s.hi, 'range'));
        api.setCaption(
          'マージ: 左 [' + s.lo + '…' + (s.mid - 1) + '] と 右 [' + s.mid + '…' + s.hi + ']'
        );
        return;
      }
      if (s.kind === 'merge_compare') {
        api.mountBars(barsEl, s.arr);
        DemoSort.assignRoles(barsEl, [
          [s.i, 'compare'],
          [s.j, 'compare'],
        ]);
        api.setCaption('比較: 位置 ' + s.i + ' と ' + s.j);
        return;
      }
      if (s.kind === 'merge_write') {
        api.mountBars(barsEl, s.arr);
        DemoSort.assignRoles(barsEl, [[s.writePos, 'write']]);
        api.setCaption('確定書き込み: 位置 ' + s.writePos);
        return;
      }
      if (s.kind === 'merge_done') {
        api.mountBars(barsEl, s.arr);
        DemoSort.clearRoles(barsEl);
        api.setCaption('区間 ' + s.lo + ' … ' + s.hi + ' のマージが完了');
        return;
      }
      if (s.kind === 'level_done') {
        api.mountBars(barsEl, s.arr);
        DemoSort.clearRoles(barsEl);
        api.setCaption('ラン幅 ' + (s.width * 2) + ' までのマージレベルが完了');
        return;
      }
      if (s.kind === 'done') {
        api.mountBars(barsEl, s.arr);
        DemoSort.clearRoles(barsEl);
        api.setCaption('ソート完了');
      }
    },
  });
});

## 類似アルゴリズムとの相違点

[ウィキソート](/2026/05/31/sort-wiki.html)は同じブロックマージ骨格を持つ。オクトソートは初期の小区間処理をオクトスワップに置き換え、回転をグリースミルズ回転にし、探索をモノバウンドにし、キャッシュ内ではクワッドマージとテイルマージを前面に出す。

[クワッドソート](/2026/08/17/sort-quad.html)は補助配列 $$O(n)$$ のボトムアップ・クワッドマージが本体である。オクトソートは小さいレベルで同様のピンポンを使い、大きいレベルではインプレース側へ切り替えて補助記憶を $$O(1)$$ に抑える。

[グレイルソート](/2026/06/01/sort-grail.html)・[コタソート](/2026/06/07/sort-kota.html)もブロックマージ系の安定インプレース整列である。オクトソートはウィキ系のバッファ戦略にクワッド系の適応マージを足した現代実装、という位置づけになる。
